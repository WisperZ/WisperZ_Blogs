## 驱动设计
基本配置信息

| -------- | ---------------------- |
| MQTT 服务器 | tcp//192.0.2.10:1884 |
| 用户名      | REPLACE_WITH_YOUR_VALUE |

采用 MQTT 驱动，采集无功补偿器的数据，具体驱动解析代码如下:
```javascript
let debug = false; // 全局调试开关

const ParseHandle = function(topic, package) {
    if (debug) console.log("topic:", topic);
    let hexString = ''; // 定义一个空字符串来存储拼接后的十六进制字符串

    package.forEach(hex => {
        const hexStr = hex.toString(16).padStart(2, '0');
        hexString += hexStr; // 将每个字符串拼接到 hexString 中
    });

    if (debug) {
        console.log("Hex Data as String:");
        console.log(hexString); // 打印完整的十六进制字符串
    }

    // 检测帧头和帧尾
    if (hexString.substring(0, 2) !== '68' || hexString.substring(hexString.length - 2) !== '16') {
        if (debug) console.log("帧头或帧尾不合格，舍弃数据帧");
        return;
    }

    // 计算校验和，从第一个68后的数据开始，到校验码之前
    let checksum = 0;
    for (let i = 12; i < hexString.length - 4; i += 2) {
        checksum += parseInt(hexString.substring(i, i + 2), 16);
    }
    checksum = checksum & 0xFF; // 只取低2位

    // 检查校验码
    const cs = parseInt(hexString.substring(hexString.length - 4, hexString.length - 2), 16);
    if (checksum !== cs) {
        if (debug) console.log("校验码不合格，舍弃数据帧");
        return;
    }

    const extractedString = topic.substring(3); // 从第4个字符开始截取到末尾
    if (debug) console.log("id:", extractedString);

    const DI_hex = hexString.substring(36, 44); 
    if (debug) console.log("DI:", DI_hex);

    if (DI_hex.toLowerCase() === '00ff0102') {
        // 解析ABC相电压数据
        const A_voltage_string = hexString.substring(46, 48) + hexString.substring(44, 46); // 97 22
        const B_voltage_string = hexString.substring(50, 52) + hexString.substring(48, 50); // 00 00
        const C_voltage_string = hexString.substring(54, 56) + hexString.substring(52, 54); // 00 00

        if (debug) {
            console.log("A相电压 String:", A_voltage_string);
            console.log("B相电压 String:", B_voltage_string);
            console.log("C相电压 String:", C_voltage_string);
        }

        // 将每个部分的十六进制数据转换为十进制，并保留一位小数
        const A_voltage = (parseInt(A_voltage_string, 10) * 0.1).toFixed(1);
        const B_voltage = (parseInt(B_voltage_string, 10) * 0.1).toFixed(1);
        const C_voltage = (parseInt(C_voltage_string, 10) * 0.1).toFixed(1);

        if (debug) {
            console.log("A相电压:", A_voltage, "V");
            console.log("B相电压:", B_voltage, "V");
            console.log("C相电压:", C_voltage, "V");
        }
        return [
            { id: extractedString, values: { A_voltage: A_voltage, B_voltage: B_voltage, C_voltage: C_voltage } }
        ];
    } else if (DI_hex.toLowerCase() === '00ff0202') {
        // 解析电流模块相关
        const A_current_string = hexString.substring(48, 50) + hexString.substring(46, 48) + hexString.substring(44, 46);
        const B_current_string = hexString.substring(54, 56) + hexString.substring(52, 54) + hexString.substring(50, 52);
        const C_current_string = hexString.substring(60, 62) + hexString.substring(58, 60) + hexString.substring(56, 58);

        if (debug) {
            console.log("A相电流 String:", A_current_string);
            console.log("B相电流 String:", B_current_string);
            console.log("C相电流 String:", C_current_string);
        }

        const A_current = (parseInt(A_current_string, 10) * 0.001).toFixed(3);
        const B_current = (parseInt(B_current_string, 10) * 0.001).toFixed(3);
        const C_current = (parseInt(C_current_string, 10) * 0.001).toFixed(3);

        if (debug) {
            console.log("A相电流:", A_current, "A");
            console.log("B相电流:", B_current, "A");
            console.log("C相电流:", C_current, "A");
        }
        return [
            { id: extractedString, values: { A_current: A_current, B_current: B_current, C_current: C_current } }
        ];
    } else if (DI_hex.toLowerCase() === '00ff0602') {
        // 总功率因数
        const total_power_factor_string = hexString.substring(46, 48) + hexString.substring(44, 46);
        const A_power_factor_string = hexString.substring(50, 52) + hexString.substring(48, 50);
        const B_power_factor_string = hexString.substring(54, 56) + hexString.substring(52, 54);
        const C_power_factor_string = hexString.substring(58, 60) + hexString.substring(56, 58);
        
        if (debug) {
            console.log("总功率因数 String:", total_power_factor_string);
            console.log("A相功率因数 String:", A_power_factor_string);
            console.log("B相功率因数 String:", B_power_factor_string);
            console.log("C相功率因数 String:", C_power_factor_string);
        }
        
        const total_power_factor = (parseInt(total_power_factor_string, 10) * 0.001).toFixed(3);
        const A_power_factor = (parseInt(A_power_factor_string, 10) * 0.001).toFixed(3);
        const B_power_factor = (parseInt(B_power_factor_string, 10) * 0.001).toFixed(3);
        const C_power_factor = (parseInt(C_power_factor_string, 10) * 0.001).toFixed(3);

        if (debug) {
            console.log("总功率因数:", total_power_factor);
            console.log("A相功率因数:", A_power_factor);
            console.log("B相功率因数:", B_power_factor);
            console.log("C相功率因数:", C_power_factor);
        }

        return [
            { id: extractedString, values: { total_power_factor: total_power_factor, A_power_factor: A_power_factor, B_power_factor: B_power_factor, C_power_factor:C_power_factor} }
        ];
    } else {
        if (debug) console.log("DI_hex does not match 00FF0102, 00FF0202, or 00FF0602. Skipping parsing.");
    }
};
```

## 数据库设计

数据库表的设计可以根据需要展示的数据和功能来进行。以下是几个关键表的设计：

1. **站点表 (stations)**
    - `station_id` (主键): 唯一标识符
    - `name`: 站点名称

2. **无功补偿器表 (compensators)**
    - `compensator_id` (主键): 唯一标识符
    - `station_id` (外键): 关联的站点 ID
    - `connectTime`: 通信时间
    - `voltage_a`: A 相电压
    - `voltage_b`: B 相电压
    - `voltage_c`: C 相电压
    - `current_a`: A 相电流
    - `current_b`: B 相电流
    - `current_c`: C 相电流
    - `A_power_factor`: A 相功率因数
    - `B_power_factor`: B 相功率因数
    - `C_power_factor`: C 相功率因数
    - `total_power_factor`: 功率因数

## 主要功能设计

1. **用户注册和登录**
    - 用户可以注册和登录，以便访问更多数据或管理数据。
    - 管理员权限设置。

2. **站点管理**
    - 添加、编辑和删除站点。
    - 查看站点详细信息和描述。

3. **无功补偿器管理**
    - 添加、编辑和删除无功补偿器。
    - 关联无功补偿器到特定站点。

4. **数据展示**
    - 实时展示各个无功补偿器的 ABC 三相电压、电流、功率因数等数据。
    - 图表展示历史数据（如折线图、柱状图等）。

5. **数据查询和过滤**
    - 根据站点位置、无功补偿器名称、时间区间等条件查询和过滤数据。

6. **告警功能**
    - 当某些参数（如电压、电流、功率因数）超过预设范围时，触发告警通知（如邮件、短信等）。
    - 告警记录和管理。
